hash.py — hash-based-threshold-signatures
def hash_lms(tweak, *data):
    h = hashlib.sha256()
    for t in tweak:
        h.update(t.to_bytes(4, "big"))
    for d in data:
        h.update(d)
    # domain-separated, tweakable hash —
    # the primitive everything else is built on
    return h.digest()
